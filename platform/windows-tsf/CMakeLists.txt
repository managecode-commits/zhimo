cmake_minimum_required(VERSION 3.22)
project(shurufa_tsf_boundary LANGUAGES CXX)

set(SHURUFA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Shurufa repository root")
add_executable(shurufa-tsf-probe core_session.cpp probe.cpp)
target_compile_features(shurufa-tsf-probe PRIVATE cxx_std_17)
target_include_directories(shurufa-tsf-probe PRIVATE "${SHURUFA_ROOT}/include")
target_link_directories(shurufa-tsf-probe PRIVATE "${SHURUFA_ROOT}/target/debug")
target_link_libraries(shurufa-tsf-probe PRIVATE ime_ffi)
if(MSVC)
  target_compile_options(shurufa-tsf-probe PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(shurufa-tsf-probe PRIVATE -Wall -Wextra -Werror)
endif()
