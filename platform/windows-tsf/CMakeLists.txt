cmake_minimum_required(VERSION 3.22)
project(shurufa_tsf_boundary LANGUAGES CXX)

set(SHURUFA_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Shurufa repository root")
if(WIN32)
  if(MSVC)
    set(default_ime_library "${SHURUFA_ROOT}/target/debug/ime_ffi.dll.lib")
    set(default_ime_runtime "${SHURUFA_ROOT}/target/debug/ime_ffi.dll")
  else()
    set(default_ime_library "${SHURUFA_ROOT}/target/x86_64-pc-windows-gnu/debug/libime_ffi.dll.a")
    set(default_ime_runtime "${SHURUFA_ROOT}/target/x86_64-pc-windows-gnu/debug/ime_ffi.dll")
  endif()
  set(SHURUFA_IME_LIBRARY "${default_ime_library}" CACHE FILEPATH "Rust Runtime import library")
  set(SHURUFA_IME_RUNTIME_DLL "${default_ime_runtime}" CACHE FILEPATH "Rust Runtime DLL")
  if(NOT EXISTS "${SHURUFA_IME_LIBRARY}")
    message(FATAL_ERROR "Build ime-ffi for the selected Windows toolchain first")
  endif()
else()
  set(SHURUFA_IME_LIBRARY ime_ffi)
endif()
add_executable(shurufa-tsf-probe core_session.cpp probe.cpp)
target_compile_features(shurufa-tsf-probe PRIVATE cxx_std_17)
target_include_directories(shurufa-tsf-probe PRIVATE "${SHURUFA_ROOT}/include")
target_link_directories(shurufa-tsf-probe PRIVATE "${SHURUFA_ROOT}/target/debug")
target_link_libraries(shurufa-tsf-probe PRIVATE "${SHURUFA_IME_LIBRARY}")
if(MSVC)
  target_compile_options(shurufa-tsf-probe PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(shurufa-tsf-probe PRIVATE -Wall -Wextra -Werror)
endif()

if(WIN32)
  add_library(shurufa-tsf SHARED core_session.cpp text_service.cpp dllmain.cpp)
  target_compile_features(shurufa-tsf PRIVATE cxx_std_17)
  target_include_directories(shurufa-tsf PRIVATE "${SHURUFA_ROOT}/include")
  target_link_directories(shurufa-tsf PRIVATE "${SHURUFA_ROOT}/target/debug")
  target_link_libraries(shurufa-tsf PRIVATE "${SHURUFA_IME_LIBRARY}" ole32 advapi32 user32 shell32)
  target_compile_definitions(shurufa-tsf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_link_libraries(shurufa-tsf PRIVATE msctf)
    target_compile_options(shurufa-tsf PRIVATE /W4 /WX /utf-8)
  else()
    target_link_libraries(shurufa-tsf PRIVATE uuid)
    target_compile_options(shurufa-tsf PRIVATE -Wall -Wextra -Werror)
  endif()
  set_target_properties(shurufa-tsf PROPERTIES OUTPUT_NAME "ShurufaTsf")
  add_custom_command(TARGET shurufa-tsf POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${SHURUFA_IME_RUNTIME_DLL}" "$<TARGET_FILE_DIR:shurufa-tsf>")
endif()
