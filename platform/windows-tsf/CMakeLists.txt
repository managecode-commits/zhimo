cmake_minimum_required(VERSION 3.22)
project(zhimo_tsf_boundary LANGUAGES CXX)

set(ZHIMO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Zhimo repository root")
if(WIN32)
  enable_language(RC)
  if(MSVC)
    set(default_ime_library "${ZHIMO_ROOT}/target/debug/ime_ffi.dll.lib")
    set(default_ime_runtime "${ZHIMO_ROOT}/target/debug/ime_ffi.dll")
  else()
    set(default_ime_library "${ZHIMO_ROOT}/target/x86_64-pc-windows-gnu/debug/libime_ffi.dll.a")
    set(default_ime_runtime "${ZHIMO_ROOT}/target/x86_64-pc-windows-gnu/debug/ime_ffi.dll")
  endif()
  set(ZHIMO_IME_LIBRARY "${default_ime_library}" CACHE FILEPATH "Rust Runtime import library")
  set(ZHIMO_IME_RUNTIME_DLL "${default_ime_runtime}" CACHE FILEPATH "Rust Runtime DLL")
  if(NOT EXISTS "${ZHIMO_IME_LIBRARY}")
    message(FATAL_ERROR "Build ime-ffi for the selected Windows toolchain first")
  endif()
else()
  set(ZHIMO_IME_LIBRARY ime_ffi)
endif()
add_executable(zhimo-tsf-probe core_session.cpp probe.cpp)
target_compile_features(zhimo-tsf-probe PRIVATE cxx_std_17)
target_include_directories(zhimo-tsf-probe PRIVATE "${ZHIMO_ROOT}/include")
target_link_directories(zhimo-tsf-probe PRIVATE "${ZHIMO_ROOT}/target/debug")
target_link_libraries(zhimo-tsf-probe PRIVATE "${ZHIMO_IME_LIBRARY}")
if(WIN32)
  target_compile_definitions(zhimo-tsf-probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(zhimo-tsf-probe PRIVATE ole32 oleaut32 user32 gdi32 uuid)
endif()
if(MSVC)
  target_compile_options(zhimo-tsf-probe PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(zhimo-tsf-probe PRIVATE -Wall -Wextra -Werror)
endif()

if(WIN32)
  add_library(zhimo-tsf SHARED core_session.cpp text_service.cpp dllmain.cpp zhimo.rc zhimo.def)
  option(ZHIMO_BUILD_DESKTOP_PANEL "Build independent handwriting/voice companion" OFF)
  if(ZHIMO_BUILD_DESKTOP_PANEL)
    set(ZHIMO_SPEECH_LIBRARY "" CACHE FILEPATH "Shared speech import library")
    if(NOT EXISTS "${ZHIMO_SPEECH_LIBRARY}")
      message(FATAL_ERROR "Build platform/shared-speech for Windows and set ZHIMO_SPEECH_LIBRARY")
    endif()
    add_executable(zhimo-desktop-panel desktop_panel.cpp zhimo.rc)
    target_compile_features(zhimo-desktop-panel PRIVATE cxx_std_17)
    target_include_directories(zhimo-desktop-panel PRIVATE "${ZHIMO_ROOT}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_definitions(zhimo-desktop-panel PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(zhimo-desktop-panel PRIVATE "${ZHIMO_IME_LIBRARY}" "${ZHIMO_SPEECH_LIBRARY}" winmm bcrypt gdi32 user32)
    if(MSVC)
      target_compile_options(zhimo-desktop-panel PRIVATE /W4 /utf-8)
    else()
      target_compile_options(zhimo-desktop-panel PRIVATE -Wall -Wextra -Werror)
    endif()
  endif()
  target_include_directories(zhimo-tsf PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
  target_compile_features(zhimo-tsf PRIVATE cxx_std_17)
  target_include_directories(zhimo-tsf PRIVATE "${ZHIMO_ROOT}/include")
  target_link_directories(zhimo-tsf PRIVATE "${ZHIMO_ROOT}/target/debug")
  # TSF interfaces are created through COM; their GUIDs are in uuid.lib.
  # The Windows SDK does not provide an msctf.lib import library.
  target_link_libraries(zhimo-tsf PRIVATE "${ZHIMO_IME_LIBRARY}" ole32 oleaut32 advapi32 user32 shell32 gdi32 uuid)
  target_compile_definitions(zhimo-tsf PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  if(MSVC)
    target_compile_options(zhimo-tsf PRIVATE /W4 /WX /utf-8)
  else()
    target_link_options(zhimo-tsf PRIVATE -Wl,--kill-at)
    target_compile_options(zhimo-tsf PRIVATE -Wall -Wextra -Werror)
  endif()
  set_target_properties(zhimo-tsf PROPERTIES OUTPUT_NAME "ZhimoTsf")
  add_custom_command(TARGET zhimo-tsf POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${ZHIMO_IME_RUNTIME_DLL}" "$<TARGET_FILE_DIR:zhimo-tsf>")
  add_custom_command(TARGET zhimo-tsf POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${ZHIMO_ROOT}/models/handwriting" "$<TARGET_FILE_DIR:zhimo-tsf>/models/handwriting")
endif()
