cmake_minimum_required(VERSION 3.22)
project(zhimo_shared_speech LANGUAGES C CXX)
set(ZHIMO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../..")
set(WHISPER_SOURCE "${ZHIMO_ROOT}/target/vendor/whisper.cpp-1.9.1" CACHE PATH "Verified whisper source")
if(NOT EXISTS "${WHISPER_SOURCE}/include/whisper.h")
  message(FATAL_ERROR "Run tools/prepare-offline-speech.sh first")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" "whisper-build")
add_library(zhimo_speech SHARED speech.cpp)
target_compile_features(zhimo_speech PRIVATE cxx_std_17)
target_compile_definitions(zhimo_speech PRIVATE ZHIMO_SPEECH_EXPORTS)
target_include_directories(zhimo_speech PUBLIC "${ZHIMO_ROOT}/include")
target_link_libraries(zhimo_speech PRIVATE whisper)
include(GNUInstallDirs)
install(TARGETS zhimo_speech LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}/zhimo"
        RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_executable(zhimo-speech-probe probe.cpp)
  target_link_libraries(zhimo-speech-probe PRIVATE zhimo_speech)
  target_compile_features(zhimo-speech-probe PRIVATE cxx_std_17)
  add_test(NAME speech-boundary COMMAND zhimo-speech-probe)
endif()
